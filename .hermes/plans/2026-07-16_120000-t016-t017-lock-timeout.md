# T-016 + T-017: SQLite Lock Contention + Task Timeout Enforcement — Implementation Plan

> **For OpenCode:** Use `opencode run --agent primary "Abarbeiten von .hermes/plans/2026-07-16_120000-t016-t017-lock-timeout.md" --thinking`

**Goal:** Add retry logic for SQLite `database is locked` errors in `db.py`, and enforce task/stage timeouts in `scheduler.py`.

**Architecture:** 
- T-016: `get_conn()` erstellt immer eine frische Connection. Bei WAL-Mode sind Reads parallel, aber konkurrierende Writes (z.B. Heartbeat + Claim gleichzeitig) können `database is locked` werfen. Ein Retry-Decorator für Write-Operationen löst das.
- T-017: `timeout_seconds` wird in `tasks` und `task_stages` gespeichert, aber nie ausgewertet. `release_expired_claims()` prüft nur `claim_expires_at` (Claim-TTL), nicht den Task-Timeout. Ein neuer `enforce_timeouts()`-Aufruf markiert Stages/Tasks als `timed_out` wenn `claimed_at + timeout_seconds < now`.

**Tech Stack:** Python, SQLite

## Task 1: Retry-Decorator für SQLite Lock Contention

**Objective:** Einen `@retry_on_locked`-Decorator in `db.py` definieren, der `database is locked`-Fehler abfängt und mit exponentiellen Backoff wiederholt.

**Files:**
- Modify: `src/relay_server/core/db.py` (Decorator + Integration in kritische Write-Funktionen)
- Modify: `tests/test_scheduler.py` (Test für Lock-Retry)

**Step 1: Decorator in db.py einfügen**

Füge nach den Imports (ca. Zeile 6) ein:

```python
import functools
import time

# ---------------------------------------------------------------------------
# Retry helper for SQLite lock contention
# ---------------------------------------------------------------------------

LOCKED_RETRIES = 5
LOCKED_BASE_DELAY = 0.05  # 50ms initial, ~1.5s total with backoff


def retry_on_locked(func):
    """Decorator: retry a DB write function on ``database is locked``.

    Uses exponential backoff (50ms → 100ms → 200ms → 400ms → 800ms).
    Raises the original ``sqlite3.OperationalError`` if all retries are
    exhausted.
    """
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        last_error = None
        delay = LOCKED_BASE_DELAY
        for attempt in range(LOCKED_RETRIES):
            try:
                return func(*args, **kwargs)
            except sqlite3.OperationalError as exc:
                msg = str(exc)
                if "database is locked" not in msg and "locked" not in msg:
                    raise
                last_error = exc
                if attempt < LOCKED_RETRIES - 1:
                    time.sleep(delay)
                    delay *= 2
        raise last_error  # type: ignore[misc]
    return wrapper
```

```python
def test_db_write_retries_on_locked():
    """Scheduler retries write when DB is locked."""
    from relay_server.core.scheduler import _retry_db_write

    call_count = 0

    @_retry_db_write
    def flaky_write():
        nonlocal call_count
        call_count += 1
        if call_count < 3:
            raise sqlite3.OperationalError("database is locked")
        return "ok"

    result = flaky_write()
    assert result == "ok"
    assert call_count == 3


def test_db_write_raises_after_exhausted_retries():
    """Scheduler raises after all retries exhausted."""
    from relay_server.core.scheduler import _retry_db_write

    call_count = 0

    @_retry_db_write
    def always_locked():
        nonlocal call_count
        call_count += 1
        raise sqlite3.OperationalError("database is locked")

    with pytest.raises(sqlite3.OperationalError, match="database is locked"):
        always_locked()
    assert call_count == 5  # _LOCKED_RETRIES
```

**Step 1: `enforce_timeouts()` in Scheduler-Klasse**

Füge nach `release_expired_claims` (ca. Zeile 331) ein:

```python
    @staticmethod
    def enforce_timeouts() -> dict[str, list[str]]:
        """Mark claimed stages and their tasks as timed_out when overdue.

        A stage is overdue when ``claimed_at + timeout_seconds < now``.
        When all stages of a task are timed_out, the task itself is also
        marked timed_out.

        Returns ``{"stages_timed_out": [...], "tasks_timed_out": [...]}``.
        """
        now = _format_time(_now())
        conn = get_conn()
        try:
            # Find overdue claimed stages.
            overdue = conn.execute(
                """
                SELECT stage_id, task_id FROM task_stages
                WHERE status = 'claimed'
                  AND datetime(claimed_at, '+' || timeout_seconds || ' seconds') < ?
                """,
                (now,),
            ).fetchall()

            timed_out_stages = [r["stage_id"] for r in overdue]
            affected_tasks: set[str] = set()

            if timed_out_stages:
                # Mark stages as timed_out.
                conn.execute(
                    """
                    UPDATE task_stages
                    SET status = 'timed_out', updated_at = ?
                    WHERE stage_id IN ({})
                    """.format(",".join("?" for _ in timed_out_stages)),
                    [now] + timed_out_stages,
                )

                # Collect affected task IDs.
                for r in overdue:
                    affected_tasks.add(r["task_id"])

                # For each affected task, check if all stages are done/timed_out.
                for task_id in affected_tasks:
                    remaining = conn.execute(
                        "SELECT COUNT(*) FROM task_stages WHERE task_id = ? AND status NOT IN ('completed', 'timed_out')",
                        (task_id,),
                    ).fetchone()[0]
                    if remaining == 0:
                        conn.execute(
                            "UPDATE tasks SET status = 'timed_out', updated_at = ?, completed_at = ? WHERE task_id = ?",
                            (now, now, task_id),
                        )

                conn.commit()

                # Publish events.
                for stage_id in timed_out_stages:
                    event_bus.publish_sync(
                        "stage_timed_out",
                        {"stage_id": stage_id},
                    )
                for task_id in affected_tasks:
                    event_bus.publish_sync(
                        "task_timed_out",
                        {"task_id": task_id},
                    )

            return {
                "stages_timed_out": timed_out_stages,
                "tasks_timed_out": list(affected_tasks),
            }
        finally:
            conn.close()
```

```python
def test_enforce_timeouts_marks_overdue_stages():
    """A claimed stage past its timeout is marked timed_out."""
    secret = _seed_admin()
    admin_id, admin_token = _register(
        secret, "Admin", [{"name": "admin", "version": "1.0"}], "admin"
    )

        secret, "Worker", [{"name": "test_ai", "version": "1.0"}], admin_token=admin_token
    )

    # Create a task with a very short timeout.
    r = client.post(
        "/relay/v2/scheduler/tasks",
        headers={"Authorization": f"Bearer {admin_token}"},
        json={
            "task_name": "timeout-test",
            "stages": [{"stage_name": "s1", "capability": "test_ai", "timeout_seconds": 1}],
        },
    )
    assert r.status_code == 200
    stage_id = r.json()["stages"][0]["stage_id"]

    # Claim the stage.
    r = client.post(
        "/relay/v2/scheduler/claim",
        headers={"Authorization": f"Bearer {worker_token}"},
        json={},
    )
    assert r.json()["claimed"] is True

    # Manually backdate claimed_at to force timeout.
    import datetime
    from relay_server.core.db import get_conn
    conn = get_conn()
    old_time = (datetime.datetime.now(datetime.timezone.utc) - datetime.timedelta(hours=1)).isoformat()
    conn.execute(
        "UPDATE task_stages SET claimed_at = ? WHERE stage_id = ?",
        (old_time, stage_id),
    )
    conn.commit()
    conn.close()

    # Enforce timeouts.
    r = client.post(
        "/relay/v2/scheduler/enforce-timeouts",
        headers={"Authorization": f"Bearer {admin_token}"},
    )
    assert r.status_code == 200
    body = r.json()
    assert stage_id in body["stages_timed_out"]

    # Verify stage is timed_out.
    r = client.get(
        f"/relay/v2/scheduler/tasks/{r.json()['tasks_timed_out'][0]}",
        headers={"Authorization": f"Bearer {admin_token}"},
    )
    stage = r.json()["task"]["stages"][0]
    assert stage["status"] == "timed_out"
```
